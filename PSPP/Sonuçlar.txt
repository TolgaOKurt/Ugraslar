objective_function(int* array, int n) {
    int max = 0;
    while (true) {
        bool bulundu = false;
        // Dizide doğrudan max + 1 var mı?
        for (int i = 0; i < n; i++) {
            if (array[i] == max + 1) {
                max++;
                bulundu = true;
                break;
            }
        }
        if (bulundu) continue;
        // Toplam ve fark kombinasyonlarını kontrol et
        for (int i = 0; i < n; i++) {
            for (int j = i; j < n; j++) {
                int toplam = array[i] + array[j];
                int fark = abs(array[i] - array[j]);
                if (toplam == max + 1 || fark == max + 1) {
                    max++;
                    bulundu = true;
                    break;
                }
            }
            if (bulundu) break;
        }
        if (!bulundu) break;
    }
    return max;
}
